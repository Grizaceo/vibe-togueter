import fs from 'node:fs';import path from 'node:path';import {execFile} from 'node:child_process';import {promisify} from 'node:util';import {VMBuilderPreflight} from '../src/vm-builder-preflight.js';
const recordName=process.argv[2]||'prewarm-control-verification.json';if(!/^prewarm-control-verification(?:-v[0-9]+)?\.json$/.test(recordName))throw Error('Exact verification record required.');
const root=path.resolve(new URL('..',import.meta.url).pathname),file=path.join(root,'runtime',recordName),run=promisify(execFile),record=JSON.parse(fs.readFileSync(file,'utf8'));
try{if(!record.closed){if(Date.now()<record.deadline)process.exit(0);const builder=new VMBuilderPreflight(path.join(root,'runtime','builder-session.json'));if(!record.name||builder.state?.name!==record.name||builder.state.requestedAt!==record.requestedAt)throw Error('Verification ownership needs reconciliation.');const ended=await builder.release();if(ended.providerTerminalStatus!=='destroyed')throw Error('Terminal destruction not confirmed.');record.closed=true;record.cleanupAt=new Date().toISOString();fs.writeFileSync(file,JSON.stringify(record)+'\n',{mode:0o600});if(builder.state.cleanupUnit)await run('systemctl',['--user','stop',builder.state.cleanupUnit+'.timer'],{timeout:10000});const device=path.join(root,'runtime','codex-vm-device-login.txt');if(fs.existsSync(device))fs.unlinkSync(device);}await run('systemctl',['--user','stop',record.unit+'.timer'],{timeout:10000});console.log('Exact control-verification cleanup confirmed.');}catch{console.error('Control verification still requires exact-session reconciliation.');process.exitCode=1;}
